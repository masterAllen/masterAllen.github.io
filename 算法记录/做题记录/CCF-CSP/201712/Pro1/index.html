Pro1

原始文件为 cpp 代码，本文是转换后的 Markdown 文件。

#include <iostream>
#include <cstring>
#include <string>
#include <cstdio>
#include <cstdlib>
#include <vector>

using namespace std;

const int maxn = 1005;
int num[maxn];
int main()
{
    //freopen("in.txt","r",stdin);
    int n;
    scanf("%d",&n);

    for (int i = 0; i < n; ++i)
        scanf("%d",&num[i]);

    int ans = 10005;
    for (int i = 0; i < n; ++i)
    {
        for (int j = i+1; j < n; ++j)
        {
            int current = num[i]-num[j];
            if (current < 0) current = -current;
            if (current < ans) ans = current;           
        }
    }

    printf("%d\n",ans);


}
